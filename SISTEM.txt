# --- 2. PANEL DEL OPERADOR (CONTROL CENTRAL) ---
class PanelOperador(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("PANEL DE CONTROL DEL OPERADOR")
        self.resize(500, 600)
        
        # Estilo oscuro moderno para cabina técnica
        self.setStyleSheet("""
            QWidget { background-color: #1e1e24; color: #e0e0e0; }
            QPushButton { 
                background-color: #2e2f38; border: 1px solid #444; 
                padding: 10px; border-radius: 5px; font-weight: bold;
            }
            QPushButton:hover { background-color: #3e404d; border: 1px solid #00bcff; }
            QListWidget { background-color: #121214; border: 1px solid #333; padding: 5px; }
        """)
        
        self.init_ui()
        self.indice_carrusel = 0

    def init_ui(self):
        layout = QVBoxLayout(self)
        
        lbl_instrucciones = QLabel("Control de Gráficos de Tiempo")
        lbl_instrucciones.setFont(QFont("Arial", 16, QFont.Weight.Bold))
        layout.addWidget(lbl_instrucciones)
        
        # Lista para seleccionar ciudades manualmente
        self.lista_ciudades = QListWidget()
        for ciudad in DATOS_CIUDADES.keys():
            self.lista_ciudades.addItem(ciudad)
        self.lista_ciudades.itemClicked.connect(self.enviar_ciudad_manual)
        layout.addWidget(self.lista_ciudades)
        
        # Botonera de control secuencial
        grid_botones = QGridLayout()
        
        self.btn_prev = QPushButton("◀ Anterior")
        self.btn_next = QPushButton("Siguiente ▶")
        self.btn_auto = QPushButton("Carrusel Automático (Play)")
        
        self.btn_prev.clicked.connect(lambda: self.navegar_secuencia(-1))
        self.btn_next.clicked.connect(lambda: self.navegar_secuencia(1))
        self.btn_auto.clicked.connect(self.conmutar_autocarrusel)
        
        grid_botones.addWidget(self.btn_prev, 0, 0)
        grid_botones.addWidget(self.btn_next, 0, 1)
        grid_botones.addWidget(self.btn_auto, 1, 0, 1, 2)
        
        layout.addLayout(grid_botones)
        
        # Timer para el modo automático de barrido país
        self.timer_carrusel = QTimer()
        self.timer_carrusel.timeout.connect(lambda: self.navegar_secuencia(1))

    def obtener_clima_api(self, ciudad_nombre):
        """Simula u obtiene datos en tiempo real de la API pública."""
        # NOTA: Reemplazar 'TU_API_KEY' por una real de OpenWeatherMap si requieres datos vivos directos.
        # Por seguridad y funcionalidad offline instantánea, genera un fallback realista estructurado.
        try:
            # Flujo de pruebas en vivo offline / simulación controlada de rangos altiplano / llanos
            import random
            es_valle_o_altiplano = any(k in ciudad_nombre for k in ["La Paz", "Oruro", "Potosí", "Vallegrande"])
            
            min_temp = random.randint(4, 11) if es_valle_o_altiplano else random.randint(18, 22)
            max_temp = min_temp + random.randint(8, 14)
            condicion = random.choice(["Clear", "Clouds", "Rain"])
            
            return {
                "icono": MAPA_ICONOS.get(condicion, "☀️"),
                "descripcion": "Despejado" if condicion == "Clear" else "Nublado" if condicion == "Clouds" else "Lluvia Aislada",
                "min": min_temp,
                "max": max_temp
            }
        except Exception:
            return {"icono": "⚠️", "descripcion": "Error de Conexión", "min": 0, "max": 0}

    def enviar_ciudad_manual(self, item):
        nombre = item.text()
        datos_clima = self.obtener_clima_api(nombre)
        comunicaciones.cambiar_ciudad.emit(nombre, datos_clima)
        # Sincronizar el índice del carrusel con el click manual
        self.indice_carrusel = list(DATOS_CIUDADES.keys()).index(nombre)

    def navegar_secuencia(self, direccion):
        ciudades = list(DATOS_CIUDADES.keys())
        self.indice_carrusel = (self.indice_carrusel + direccion) % len(ciudades)
        
        nombre = ciudades[self.indice_carrusel]
        self.lista_ciudades.setCurrentRow(self.indice_carrusel)
        datos_clima = self.obtener_clima_api(nombre)
        comunicaciones.cambiar_ciudad.emit(nombre, datos_clima)

    def conmutar_autocarrusel(self):
        if self.timer_carrusel.isActive():
            self.timer_carrusel.stop()
            self.btn_auto.setText("Carrusel Automático (Play)")
            self.btn_auto.setStyleSheet("")
        else:
            self.timer_carrusel.start(4000) # Cambia cada 4 segundos recorriendo el país
            self.btn_auto.setText("Detener Carrusel (Pause)")
            self.btn_auto.setStyleSheet("background-color: #d32f2f; color: white;")
